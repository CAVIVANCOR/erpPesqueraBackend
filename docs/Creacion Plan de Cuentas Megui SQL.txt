-- ============================================
-- CONSULTA PLAN CONTABLE CON DATOS DE CUENTA PADRE
-- ============================================
SELECT 
    pc.id,
    pc."codigoCuenta",
    pc."nombreCuenta",
    pc.descripcion,
    pc.nivel,
    pc."cuentaPadreId",
    padre."codigoCuenta" AS "codigoCuentaPadre",
    padre."nombreCuenta" AS "nombreCuentaPadre",
    pc.naturaleza,
    pc."esImputable",
    pc."requiereCentroCosto",
    pc."requiereEntidad",
    pc."requiereProyecto",
    pc."tipoCuenta",
    pc."esActivoCorriente",
    pc."esActivoNoCorriente",
    pc."esPasivoCorriente",
    pc."esPasivoNoCorriente",
    pc.activo,
    pc."creadoEn",
    pc."creadoPor",
    pc."actualizadoEn",
    pc."actualizadoPor"
FROM 
    "PlanCuentasContable" pc
LEFT JOIN 
    "PlanCuentasContable" padre ON pc."cuentaPadreId" = padre.id
ORDER BY 
    pc."codigoCuenta" ASC;




-- ============================================
-- SCRIPT MEJORADO PARA MAYÚSCULAS SIN ACENTOS
-- PLAN CONTABLE GENERAL EMPRESARIAL (PCGE)
-- ============================================

UPDATE "PlanCuentasContable"
SET 
    "nombreCuenta" = UPPER(
        REGEXP_REPLACE(
            REGEXP_REPLACE(
                REGEXP_REPLACE(
                    REGEXP_REPLACE(
                        REGEXP_REPLACE("nombreCuenta", '[áàâãäå]', 'a', 'gi'),
                    '[éèêë]', 'e', 'gi'),
                '[íìîï]', 'i', 'gi'),
            '[óòôõö]', 'o', 'gi'),
        '[úùûü]', 'u', 'gi')
    ),
    descripcion = UPPER(
        REGEXP_REPLACE(
            REGEXP_REPLACE(
                REGEXP_REPLACE(
                    REGEXP_REPLACE(
                        REGEXP_REPLACE(descripcion, '[áàâãäå]', 'a', 'gi'),
                    '[éèêë]', 'e', 'gi'),
                '[íìîï]', 'i', 'gi'),
            '[óòôõö]', 'o', 'gi'),
        '[úùûü]', 'u', 'gi')
    ),
    "actualizadoEn" = NOW()
WHERE activo = true;

    SELECT id 
    FROM "PlanCuentasContable"

WHERE "cuentaContableId" IS NULL;
